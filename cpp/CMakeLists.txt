cmake_minimum_required(VERSION 3.16)
project(AruRetopoCore LANGUAGES CXX)
add_library(aru_retopo_core SHARED retopo.cpp)
set_target_properties(aru_retopo_core PROPERTIES OUTPUT_NAME aru_retopo_core_v5)
target_compile_features(aru_retopo_core PRIVATE cxx_std_17)
if(MSVC)
  target_compile_options(aru_retopo_core PRIVATE /O2 /W4 /EHsc)
else()
  target_compile_options(aru_retopo_core PRIVATE -O3 -Wall -Wextra)
endif()
install(TARGETS aru_retopo_core RUNTIME DESTINATION bin LIBRARY DESTINATION bin)

add_library(aru_retopo_display SHARED display.cpp)
set_target_properties(aru_retopo_display PROPERTIES OUTPUT_NAME aru_retopo_display_v3)
target_compile_features(aru_retopo_display PRIVATE cxx_std_17)
install(TARGETS aru_retopo_display RUNTIME DESTINATION bin LIBRARY DESTINATION bin)

find_package(OpenMP)
if(OpenMP_CXX_FOUND)
  target_link_libraries(aru_retopo_core PRIVATE OpenMP::OpenMP_CXX)
endif()
